/* Write value as digit_count right-aligned seven-segment digits to output.
 * With display_mode 1, leading zeros are blanked (the units digit is always
 * shown). */
#include <stdint.h>
#include "firmware/reconstructed/include/sr08_display_value_formatters.h"

enum {
    MAX_DIGITS = 8,
    BLANK = 16,
};

/* Segment patterns for 0-9, A-F and a blank digit. */
static const uint8_t segment_patterns[17] = {
    0x77, 0x24, 0x5d, 0x6d, 0x2e, 0x6b, 0x7b, 0x25, 0x7f, 0x6f, 0x3f, 0x3a, 0x53, 0x7c, 0x5b, 0x1b, 0x00,
};

void convert_quotient_remainder_sequence_ring_s_at_3148(uint32_t value, volatile uint8_t *output, uint32_t digit_count,
                                                        uint32_t display_mode)
{
    uint8_t digits[MAX_DIGITS];
    uint32_t place = 1;
    uint32_t i;

    for (i = 0; i < digit_count; i++) {
        uint8_t digit = value / place % 10;
        uint8_t shown = place > value ? BLANK : digit;

        if (i - 1 >= i)
            shown = digit;
        if (display_mode != 1)
            shown = digit;
        digits[digit_count - 1 - i] = shown;
        place *= 10;
    }
    for (i = 0; i < digit_count; i++)
        output[i] = segment_patterns[digits[i]];
}
