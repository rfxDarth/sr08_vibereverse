def validate_compiler_configuration(manifest: dict) -> None:

    symbols = {row["symbol"] for row in manifest["sources"]}

                  | SDK_COVERED_UNITS | ARM_COMPILER_UNITS | ARM_LIBRARY_UNITS | SDK_ASSEMBLY_UNITS)

def read_source_units(manifest: dict) -> list[tuple[dict, Path, str]]:
    """Read the manifested code units and reject duplicate section addresses."""
    units = [{"address": f"0x{item['address']:08x}",
              "function": item["symbol"], "kind": item["kind"],
              "source": item["source"]} for item in manifest["sources"]]
    addresses = [int(unit["address"], 16) for unit in units]
    if len(addresses) != len(set(addresses)):
        raise SystemExit("duplicate exact function starts in source manifests")
    source_root = ROOT / "firmware" / "reconstructed"
    source_units = []
    for unit in units:
        source = (ROOT / unit["source"]).resolve()
        if not source.is_relative_to(source_root.resolve()):
            raise SystemExit(f"manifest source is outside reconstruction tree: {unit['source']}")
        source_units.append((unit, source, source.read_text(errors="replace")))
    return source_units


def resolve_linker_symbols(manifest: dict, source_units: list[tuple[dict, Path, str]]) -> dict[str, int]:
    """Resolve source references to fixed firmware and ROM addresses."""
    function_addresses = {item["symbol"]: item["address"] for item in manifest["sources"]}
    aliases = manifest["function_aliases"]
    conflicts = {name for name, address in aliases.items()
                 if name in function_addresses and function_addresses[name] != address}
    if conflicts:
        raise SystemExit(f"function aliases conflict with source entries: {', '.join(sorted(conflicts))}")
    function_addresses.update(aliases)
    rom_addresses = manifest["rom_symbols"]
    referenced_functions: set[str] = set()
    referenced_rom: set[str] = set()
    identifiers: set[str] = set()
    for _, _, code in source_units:
        clean = COMMENT_RE.sub("", code)
        names = set(IDENTIFIER_RE.findall(clean))
        referenced_functions.update(names & function_addresses.keys())
        referenced_rom.update(f"func_0x{address.lower()}" for address in ROM_RE.findall(clean))
        identifiers.update(names)

    definitions = {name: function_addresses[name] for name in referenced_functions}
    definitions.update({name: int(name.rsplit("_", 1)[1], 16) for name in referenced_rom})
    definitions.update({name: rom_addresses[name] for name in identifiers & rom_addresses.keys()})
    return definitions


def compile_source(cc: str, source: Path, output: Path, *, contents: str | None = None,
                   description: str | None = None, flags: list[str] = FLAGS) -> None:
    """Compile one source file with the exact-build target and warning policy."""
    if contents is not None:
        source.write_text(contents)
    build = subprocess.run([cc, *flags, "-c", str(source), "-o", str(output)],
                           capture_output=True, text=True)
    if build.returncode:
        label = description or str(source)
        raise SystemExit(f"compile failed for {label}:\n{build.stderr[-2000:]}")

def lto_regions(func_starts: list[int], addresses: set[int]) -> list[tuple[int, int]]:
    """Each LTO function's V204 region, up to the next function section (its pool included),
    and each LTO read-only object's, up to the next layout section."""
    ends = func_starts + [BASE + IMAGE_SIZE]
    regions = [(address, next(end for end in ends if end > address)) for address in sorted(addresses)]
    sections = sorted(layout_input_sections()) + [BASE + IMAGE_SIZE]
    regions += [(address, next(end for end in sections if end > address))
                for _, address, _, _ in LTO_READONLY_SOURCES]
    return sorted(regions)


def loads_into(image: bytes, start: int, end: int, regions: list[tuple[int, int]]) -> bool:
    """Whether a Thumb literal load in [start, end) of ``image`` points into one of ``regions``."""
    for address in range(start & ~1, end - 1, 2):
        half = int.from_bytes(image[address - BASE:address - BASE + 2], "little")
        if (half & 0xF800) == 0x4800:
            target = ((address + 4) & ~3) + (half & 0xFF) * 4
            if any(low <= target < high for low, high in regions):
                return True
    return False


def lto_context_pools(image: bytes, func_starts: list[int], exclusions: list[tuple[int, int]]) -> list[tuple[int, bytes]]:
    """The code and pools within 1 KiB after each LTO region, as (address, bytes) stubs.

    armlink retargets a literal load to an identical word in a later in-range
    pool, so the LTO module only links as in V204 when those surroundings take
    part in the link. Earlier code that loads a literal from a module region
    takes part as well, so that armlink keeps that literal. ``image`` is the payload; ``exclusions`` are the module's
    regions. Function boundaries are preserved: combining unrelated pools into
    one input section permits armlink to delete pool words.
    """
    boundaries = {BASE, BASE + IMAGE_SIZE}
    boundaries.update(address for address in func_starts if address % 4 == 0)
    for start, end in exclusions:
        boundaries.update((start, end))
    boundaries = sorted(boundaries)
    pools = []
    for start, end in zip(boundaries, boundaries[1:]):
        if any(start < high and low < end for low, high in exclusions):
            continue
        if not any(high <= start < high + 1024 for _, high in exclusions) and \
                not loads_into(image, start, end, exclusions):
            continue
        aligned_start = (start + 3) & ~3
        aligned_end = end & ~3
        if aligned_end > aligned_start:
            pools.append((aligned_start, bytes(image[aligned_start - BASE:aligned_end - BASE])))
    return pools

def place_arm_compiler_groups(compilers: arm_compilers.ArmCompilers, objcopy: str, temp: Path,
                              source_units: list[tuple[dict, Path, str]],
                              symbols: dict[str, int]) -> list[arm_compilers.PlacedGroup]:
    """Compile and link each Arm compiler group at its V204 address."""
    units = {row["function"]: (int(row["address"], 16), code) for row, _, code in source_units}
    if len(ARM_COMPILER_UNITS) != sum(len(group) for _, group in ARM_COMPILER_GROUPS):
        raise SystemExit("a source unit appears in more than one Arm compiler group")
    placed = []
    lto_module = []
    for compiler, group in ARM_COMPILER_GROUPS:
        members = [(symbol, *units[symbol]) for symbol in group]
        if compiler in arm_compilers.LTO_COMPILERS:
            lto_module.extend(members)
            continue
        if [address for _, address, _ in members] != sorted(address for _, address, _ in members):
            raise SystemExit(f"Arm compiler group {group[0]} is not in address order")
        placed.append(compilers.place_group(compiler, members, ROOT, temp, symbols, set(ram_symbols()) | FIXED_DATA_SYMBOLS.keys(),
                                            objcopy, "arm-none-eabi-readelf"))
    # The LTO modules are linked later, once the surrounding code exists (see main).
    return placed

def place_arm_library_members(tools_dir: Path, objcopy: str, temp: Path, manifest: dict
                              ) -> tuple[list[arm_compilers.PlacedGroup], dict[str, int]]:
    """Extract the microlib members V204 linked and return their ranges and globals."""
    selectors = layout_input_sections()
    addresses = {row["symbol"]: row["address"] for row in manifest["sources"]}
    if len(ARM_LIBRARY_UNITS) != sum(len(units) for _, _, units in ARM_LIBRARY_MEMBERS) \
            or ARM_LIBRARY_UNITS & ARM_COMPILER_UNITS:
        raise SystemExit("a source unit is replaced more than once")
    placed, exported = [], {}
    for start, members, units in ARM_LIBRARY_MEMBERS:

        if not all(group.start <= addresses[unit] < group.end for unit in units):

def place_sdk_assembly(compilers: arm_compilers.ArmCompilers, objcopy: str, temp: Path, manifest: dict,
                       fixed: dict[str, int]) -> list[arm_compilers.PlacedGroup]:
    """Assemble the official SDK assembly sources V204 used (SDK_ASSEMBLY_SOURCES)."""
    addresses = {row["symbol"]: row["address"] for row in manifest["sources"]}
    if SDK_ASSEMBLY_UNITS & (ARM_COMPILER_UNITS | ARM_LIBRARY_UNITS | SDK_UNITS.keys() | SDK_COVERED_UNITS):
        raise SystemExit("an SDK assembly source replaces a unit that is already replaced")
    placed = []
    for start, source, digest, options, entry, units in SDK_ASSEMBLY_SOURCES:
        if hashlib.sha256(source.read_bytes()).hexdigest() != digest:
            raise SystemExit(f"{source} differs from the verified SDK file")
        placed.append(compilers.place_assembly(source, options, entry, [(unit, addresses[unit]) for unit in units],
                                               start, temp, fixed, objcopy, "arm-none-eabi-readelf"))
    return placed


def sdk_rom_symbols() -> dict[str, tuple[int, bool]]:

def verified_sdk_archive(archive: Path) -> Path:
    """Return ``archive`` after checking it against its recorded SHA-256."""
    if hashlib.sha256(archive.read_bytes()).hexdigest() != SDK_ARCHIVE_HASHES[archive]:
        raise SystemExit(f"{archive} differs from the verified SDK library")
    return archive


def place_sdk_members(objcopy: str, temp: Path, manifest: dict, fixed: dict[str, int]
                      ) -> tuple[list[arm_compilers.PlacedGroup], set[str]]:
    """Link official SDK library members at their V204 addresses (SDK_MEMBERS)."""
    selectors = layout_input_sections()
    rom = sdk_rom_symbols()
    addresses = {row["symbol"]: row["address"] for row in manifest["sources"]}
    functions = addresses.keys() | manifest["function_aliases"].keys()
    sources = {row["symbol"]: row["source"] for row in manifest["sources"]}
    placed, replaced = [], set()
    for archive, member, sections in SDK_MEMBERS:
        stem = f"sdk_{archive.parent.name}_{member}"
        obj = temp / stem
        obj.write_bytes(subprocess.check_output(["arm-none-eabi-ar", "p", str(verified_sdk_archive(archive)),
                                                 member]))
        headers = subprocess.run(["arm-none-eabi-readelf", "-SW", str(obj)], check=True,
                                 capture_output=True, text=True).stdout
        # Out-of-line copies of ROM helpers (armcc's i.<name> sections) go to the
        # helper's ROM address, so calls to them resolve as V204's do.
        helpers = re.findall(r"^\s*\[\s*\d+\]\s+i\.(\S+)\s+PROGBITS", headers, re.M)
        table = subprocess.run(["arm-none-eabi-readelf", "-sW", str(obj)], check=True,
                               capture_output=True, text=True).stdout
        undefined = set(re.findall(r"^\s*\d+:\s+0+\s+\d+\s+\w+\s+\w+\s+\w+\s+UND\s+(\S+)$", table, re.M))
        relocations = subprocess.run(["arm-none-eabi-readelf", "-rW", str(obj)], check=True,
                                     capture_output=True, text=True).stdout
        referenced = set(re.findall(r"^[0-9a-f]{8}\s+[0-9a-f]+\s+\S+\s+[0-9a-f]+\s+(\S+)", relocations, re.M))
        defsyms = []
        for name in sorted(undefined):
            target = arm_compilers.ROM_RUNTIME_ALIASES.get(name, name)
            if target in fixed:
                # Thumb code addresses carry bit zero in data pointers.
                address = fixed[target] | (target in functions or rom.get(target, (0, False))[1])
            elif target in rom:
                address = rom[target][0] | rom[target][1]
            elif name in referenced:
                raise SystemExit(f"{member}: unresolved reference to {name}")
            else:
                continue
            defsyms.append(f"--defsym={name}=0x{address:x}")
        script = [f"{section} 0x{address:x} : {{ *({section}) }}" for section, address in sections.items()]
        script += [f"i.{name} 0x{rom[name][0]:x} : {{ *(i.{name}) }}" for name in helpers]
        layout = temp / f"{stem}.ld"
        layout.write_text("SECTIONS { " + " ".join(script) + " /DISCARD/ : { *(*) } }\n")
        image = temp / f"{stem}.elf"
        subprocess.run(["arm-none-eabi-ld", "-T", str(layout), *defsyms, str(obj), "-o", str(image)], check=True)
        for section, start in sections.items():
            raw = temp / f"{stem}{section}.bin"
            subprocess.run([objcopy, "-O", "binary", "-j", section, str(image), str(raw)], check=True)
            end = start + raw.stat().st_size
            units = sorted((symbol for symbol, address in addresses.items() if start <= address < end),
                           key=addresses.get)
            code = section == ".text"
            if code and (not units or addresses[units[0]] != start):
                raise SystemExit(f"{member} {section} at 0x{start:08x} must start at a source unit")
            if not code and units:
                raise SystemExit(f"{member} {section} covers source units {units}")
            selector = selectors[start]
            wrapped = temp / f"{stem}{section}.o"
            flags = "alloc,load,readonly," + ("code," if code else "data,") + "contents"
            subprocess.run([objcopy, "-I", "binary", "-O", "elf32-littlearm", "-B", "arm",
                            "--rename-section", f".data={selector},{flags}", raw.name, str(wrapped)],
                           check=True, cwd=temp)
            local_symbols = [f"{unit}={selector}:0x{addresses[unit] - start:x},function,local" for unit in units]
            # Inline assembly in other units may still branch to address-named
            # labels inside the range; define them at those addresses.
            labels = set()
            for symbol, source in sources.items():
                if symbol not in units:
                    labels.update(re.findall(r"\b(\w+_(07f[0-9a-f]{5}))\b", (ROOT / source).read_text()))
            for label, address in sorted(labels):
                if start <= int(address, 16) < end and label not in addresses:
                    local_symbols.append(f"{label}={selector}:0x{int(address, 16) - start:x},function,global")
            marker = ["--add-symbol", f"$t={selector}:0,local"] if code else []
            subprocess.run([objcopy, "--strip-all", *marker,
                            *[arg for item in local_symbols for arg in ("--add-symbol", item)], str(wrapped)],
                           check=True)
            arm_compilers.mark_eabi5(wrapped)
            placed.append(arm_compilers.PlacedGroup(units, start, end, wrapped))
            replaced.update(units)
    return placed, replaced


def covered_by(placed: list[arm_compilers.PlacedGroup], start: int, size: int, label: str) -> bool:
    """Report whether a V204 range lies inside a placed group; reject partial overlap."""
    for group in placed:
        if group.start <= start and start + size <= group.end:
            return True
        if start < group.end and group.start < start + size:
            raise SystemExit(f"{label} partially overlaps compiled group {group.symbols[0]} "
                             f"(0x{group.start:08x}-0x{group.end:08x})")
    return False


def compile_source_units(cc: str, gcc: str, temp: Path,
                         source_units: list[tuple[dict, Path, str]],
                         placed: list[arm_compilers.PlacedGroup], sdk_slice_units: set[str]) -> list[Path]:
    """Compile manifested functions and fixed data into temporary objects."""
    objects = [group.obj for group in placed]
    native_ranges = []
    for number, (row, source, code) in enumerate(source_units):
        cfile = temp / f"function_{number:03}.c"
        obj = temp / f"function_{number:03}.o"
        if row["function"] in (SDK_COVERED_UNITS | ARM_COMPILER_UNITS | ARM_LIBRARY_UNITS | SDK_ASSEMBLY_UNITS
                               | sdk_slice_units):
            continue
        if covered_by(placed, int(row["address"], 16), 1, f"source unit {row['function']}"):
            raise SystemExit(f"source unit {row['function']} lies inside a placed Arm group "
                             "but is not a member of it")
        if row["function"] in SDK_UNITS:
            obj.write_bytes(subprocess.check_output(
                ["arm-none-eabi-ar", "p", str(verified_sdk_archive(SDK_ARCHIVE)), SDK_UNITS[row["function"]]]))
            subprocess.run(["arm-none-eabi-objcopy", "--strip-debug", "--rename-section",
                            ".text=.text." + row["function"], str(obj)], check=True)
            objects.append(obj)
            continue
        use_gcc = row["function"] in GCC_TRANSLATION_UNITS
        compile_source(gcc if use_gcc else cc, cfile, obj, contents=code,
                       description=f"{row['address']} ({source})",
                       flags=(GCC_FLAGS if use_gcc else FLAGS) + SOURCE_FLAGS.get(row["function"], []))
        objects.append(obj)
        # Native C may emit its own literal pool inside the function section.
        # Treat that emitted range like an Arm compiler group when removing
        # separately maintained data that it now owns. Partial overlaps still
        # fail, and the complete image must pass the canonical hash check.
        section_headers = subprocess.run(["arm-none-eabi-objdump", "-h", str(obj)],
                                         check=True, capture_output=True, text=True).stdout
        text_size = re.search(r"^\s*\d+\s+\.text\." + re.escape(row["function"])
                              + r"\s+([0-9a-fA-F]+)\s", section_headers, re.M)
        if text_size:
            start = int(row["address"], 16)
            native_ranges.append(arm_compilers.PlacedGroup(
                [row["function"]], start, start + int(text_size.group(1), 16), obj))

    data_owners = placed + native_ranges
    for number, source in enumerate((DATA_DIR / "V204_tables.c",
                                     DATA_DIR / "V204_auto_data.c")):
        obj = temp / f"data_{number}.o"
        compile_source(cc, source, obj)
        headers = subprocess.run(["arm-none-eabi-objdump", "-h", str(obj)], check=True,
                                 capture_output=True, text=True).stdout
        replaced = list(SDK_REPLACED_DATA_SECTIONS)
        for name, size_hex, _ in DATA_SECTION_RE.findall(headers):
            if covered_by(data_owners, int(name.rsplit(".", 1)[1], 16), int(size_hex, 16),
                          f"data section {name}"):
                replaced.append(name)
        subprocess.run(["arm-none-eabi-objcopy",
                        *["--remove-section=" + name for name in replaced],
                        str(obj)], check=True)
        objects.append(obj)
    return objects


def link_source_objects(ld: str, objdump: str, output: Path, temp: Path,
                        objects: list[Path], linker_symbols: dict[str, int]) -> list[tuple[str, str, str]]:
    """Link exact sections at their fixed VMAs and return the section map."""
    linker_script = temp / "V204_data_sections.ld"
    linker_script.write_text((DATA_DIR / "V204_data_sections.ld").read_text())
    command = [ld, "-T", str(linker_script)]
    command.extend(f"--defsym={name}=0x{address:x}"
                   for name, address in sorted(linker_symbols.items()))
    command.extend(str(obj) for obj in objects)
    command.extend(["-o", str(output)])
    link = subprocess.run(command, capture_output=True, text=True)
    if link.returncode:
        raise SystemExit(f"link failed:\n{link.stderr[-4000:]}")

    headers = subprocess.run([objdump, "-h", str(output)], check=True,
                             capture_output=True, text=True).stdout
    sections = SECTION_RE.findall(headers)
    if not sections:
        raise SystemExit("no V204 exact sections in linked ELF")
    return sections


def reconstruct_payload(objcopy: str, output: Path, temp: Path,
                        sections: list[tuple[str, str, str]],
                        reference_payload: bytes | None) -> tuple[bytearray, int]:
    """Extract linked sections, verify their layout, and assemble the payload."""
    coverage = bytearray(IMAGE_SIZE)
    reconstructed = bytearray(IMAGE_SIZE)
    for name, size_hex, vma_hex in sections:
        address, size, vma = int(name.rsplit(".", 1)[1], 16), \
                             int(size_hex, 16), int(vma_hex, 16)
        if address != vma:
            raise SystemExit(f"{name}: VMA 0x{vma:x} != address 0x{address:x}")
        if address < BASE or address + size > BASE + IMAGE_SIZE:
            raise SystemExit(f"{name}: outside V204 payload")
        offset = address - BASE
        end = offset + size
        if any(coverage[offset:end]):
            raise SystemExit(f"overlapping linked sections at {name}")
        coverage[offset:end] = b"\x01" * size

        section_file = temp / f"{address:08x}.bin"
        subprocess.run([objcopy, "-O", "binary", "-j", name,
                        str(output), str(section_file)], check=True)
        actual = section_file.read_bytes()
        if reference_payload is not None:
            expected = reference_payload[offset:end]
            if actual != expected:
                raise SystemExit(f"{name}: bytes differ from V204 payload")
        reconstructed[offset:end] = actual

    covered = sum(coverage)
    if covered != IMAGE_SIZE:
        raise SystemExit(f"linked payload is incomplete: {IMAGE_SIZE - covered} "
                         "payload bytes are uncovered")
    if reference_payload is not None and reconstructed != reference_payload:
        raise SystemExit("reconstructed payload differs from reference payload")
    payload_hash = hashlib.sha256(reconstructed).hexdigest()
    if payload_hash != CANONICAL_PAYLOAD_SHA256:
        raise SystemExit(f"payload SHA-256 {payload_hash} does not match canonical V204")
    return reconstructed, covered


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--cc", default="clang-18")
    parser.add_argument("--gcc", default="arm-none-eabi-gcc",
                        help="GNU Arm compiler for C units pinned to its code generation")
    parser.add_argument("--arm-tools-dir", type=Path, default=ARM_TOOLS_DIR,
                        help="directory with armcc506/, armclang616/, and mdk_lite/ (see arm_compilers.py)")
    parser.add_argument("--ld", default="arm-none-eabi-ld")
    parser.add_argument("--objcopy", default="arm-none-eabi-objcopy")
    parser.add_argument("--objdump", default="arm-none-eabi-objdump")
    parser.add_argument("--output", type=Path,
                        default=Path(tempfile.gettempdir()) / "V204_exact.elf",
                        help="write the linked ELF here (default: system temp directory)")
    parser.add_argument("--flat-output", type=Path,
                        help="write a raw image from linked sections; requires complete coverage")
    parser.add_argument("--container-output", type=Path,
                        help="prepend the source-defined 64-byte SUOTA header")
    parser.add_argument("--verify-payload", type=Path,
                        help="optionally compare linked bytes against a reference payload")
    parser.add_argument("--verify-container", type=Path,
                        help="optionally compare the generated container against a reference image")
    args = parser.parse_args()
    reference_payload = args.verify_payload.read_bytes() if args.verify_payload else None
    if reference_payload is not None and len(reference_payload) != IMAGE_SIZE:
        raise SystemExit(f"reference payload must be {IMAGE_SIZE} bytes")
    manifest = json.loads(MANIFEST.read_text())
    if manifest.get("format") != 2:
        raise SystemExit("unsupported source manifest format")
    if manifest["image_base"] != BASE or manifest["image_size"] != IMAGE_SIZE:
        raise SystemExit("source manifest flash map does not match the linker map")
    validate_compiler_configuration(manifest)
    source_specs = read_source_units(manifest)
    functions = [row for row, _, _ in source_specs]
    defs = resolve_linker_symbols(manifest, source_specs)
    defs.update(manifest["rom_symbols"])
    defs.update(SDK_LINKER_SYMBOLS)
    defs.update(FIXED_DATA_SYMBOLS)

    args.output.parent.mkdir(parents=True, exist_ok=True)
    with tempfile.TemporaryDirectory(prefix="v204-source-build-") as directory:
        temp = Path(directory)
        placed, library_symbols = place_arm_library_members(args.arm_tools_dir, args.objcopy, temp, manifest)
        fixed = all_fixed_symbols(manifest)
        fixed.update(library_symbols)
        fixed.update(ARM_LIBRARY_LINKER_SYMBOLS)
        placed += place_sdk_assembly(arm_compilers.ArmCompilers(args.arm_tools_dir), args.objcopy, temp,
                                     manifest, fixed)
        if ARM_COMPILER_GROUPS:
            placed += place_arm_compiler_groups(arm_compilers.ArmCompilers(args.arm_tools_dir),
                                                args.objcopy, temp, source_specs, fixed)
        defs.update(ARM_LIBRARY_LINKER_SYMBOLS)
        # Aliases name Thumb functions; library literals store them with bit zero set.
        defs.update({name: fixed[target] | 1 for name, target in ARM_LIBRARY_SYMBOL_ALIASES.items()})
        for name in library_symbols:
            defs.pop(name, None)
        slices, sdk_slice_units = place_sdk_members(args.objcopy, temp, manifest, fixed)
        if sdk_slice_units & (SDK_UNITS.keys() | SDK_COVERED_UNITS | ARM_COMPILER_UNITS | ARM_LIBRARY_UNITS):
            raise SystemExit("an SDK library member replaces a unit that is already replaced")
        placed += slices
        objects = compile_source_units(args.cc, args.gcc, temp, source_specs, placed, sdk_slice_units)
        sections = link_source_objects(args.ld, args.objdump, args.output,
                                       temp, objects, defs)
        # The first link leaves out the LTO module (its functions are fixed
        # symbols). armlink then gets the source-built surrounding code and
        # pools, just as the original whole-image link did, and links the
        # module; the final link adds it. No firmware reference supplies bytes.
        # A second module ("ac6ltowordtables") links the same way once the first
        # one's code is in the image; regions of modules not linked yet take no
        # part as context.
        for lto_compiler in arm_compilers.LTO_COMPILERS:
            lto_symbols = {symbol for compiler, group in ARM_COMPILER_GROUPS
                           if compiler == lto_compiler for symbol in group}
            if not lto_symbols:
                continue
            main_module = lto_compiler == "ac6lto"
            later_symbols = {symbol for compiler, group in ARM_COMPILER_GROUPS
                             if compiler in arm_compilers.LTO_COMPILERS[
                                 arm_compilers.LTO_COMPILERS.index(lto_compiler) + 1:]
                             for symbol in group}
            module_temp = temp if main_module else temp / lto_compiler
            module_temp.mkdir(exist_ok=True)
            context = bytearray(IMAGE_SIZE)
            for name, size, address in sections:
                raw_pool = temp / (name + ".context.bin")
                subprocess.run([args.objcopy, "-O", "binary", "-j", name,
                                str(args.output), str(raw_pool)], check=True)
                start = int(address, 16) - BASE
                context[start:start + int(size, 16)] = raw_pool.read_bytes()
            func_starts = layout_function_starts()
            exclusions = lto_regions(func_starts, {int(row["address"], 16) for row, _, _ in source_specs
                                                   if row["function"] in lto_symbols | later_symbols})
            pools = lto_context_pools(context, func_starts, exclusions)
            module = [(row["function"], int(row["address"], 16), code)
                      for row, _, code in source_specs if row["function"] in lto_symbols]
            if main_module:
                check_lto_internal(source_specs, lto_symbols)
            rebuilt = arm_compilers.ArmCompilers(args.arm_tools_dir).place_lto_module(
                module, lto_support_sources() if main_module else [], ROOT, module_temp,
                *lto_link_symbols(fixed),
                args.objcopy, "arm-none-eabi-readelf", pools, layout_pool_words(),
                readonly_units=lto_readonly_sources() if main_module else [],
                context_word=lambda address: int.from_bytes(context[address - BASE:address - BASE + 4], "little"),
                internal=LTO_INTERNAL_UNITS if main_module else set(),
                native_units=LTO_NATIVE_UNITS if main_module else set(),
                sealed_literals=LTO_SEALED_STUB_LITERALS if main_module else (),
                separate_code_starts=frozenset(int(row["address"], 16)
                                              for row, _, _ in source_specs
                                              if row["function"] in later_symbols),
                compiler=lto_compiler)
            # Recompile without the data sections the module's regions now cover.
            placed += rebuilt
            objects = compile_source_units(args.cc, args.gcc, temp, source_specs, placed, sdk_slice_units)
            sections = link_source_objects(args.ld, args.objdump, args.output, temp, objects, defs)
        reconstructed, covered = reconstruct_payload(
            args.objcopy, args.output, temp, sections, reference_payload)
        if args.flat_output:
            args.flat_output.parent.mkdir(parents=True, exist_ok=True)
            args.flat_output.write_bytes(reconstructed)
            print(f"flat image: {args.flat_output} ({len(reconstructed)} bytes)")
        if args.container_output or args.verify_container:
            header_source = DATA_DIR / "V204_ota_header.c"
            header_obj = temp / "v204_ota_header.o"
            compile_source(args.cc, header_source, header_obj)
            header_path = temp / "v204_ota_header.bin"
            subprocess.run([args.objcopy, "-O", "binary", "-j", ".v204.header",
                            str(header_obj), str(header_path)], check=True)
            header = header_path.read_bytes()
            if len(header) != 64:
                raise SystemExit("compiled C container header is not 64 bytes")
            container = header + reconstructed
            if args.verify_container:
                original = args.verify_container.read_bytes()
                if container != original:
                    raise SystemExit("reconstructed OTA container differs from reference image")
            container_hash = hashlib.sha256(container).hexdigest()
            if container_hash != CANONICAL_CONTAINER_SHA256:
                raise SystemExit(f"container SHA-256 {container_hash} does not match canonical V204")
            if args.container_output:
                args.container_output.parent.mkdir(parents=True, exist_ok=True)
                args.container_output.write_bytes(container)
                print(f"OTA container: {args.container_output} ({len(container)} bytes)")
        if reference_payload is not None:
            verification = "verified against reference and canonical hash"
        else:
            verification = "matches canonical hash"
        print(f"linked {len(sections)} sections, {covered} bytes "
              f"({100 * covered / IMAGE_SIZE:.2f}% of payload; {verification}); "
              f"{len(functions)} source units")
        print(f"ELF: {args.output}")


if __name__ == "__main__":
    main()
