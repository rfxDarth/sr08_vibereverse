/* Take over a fresh four-value sensor reading (first value neither 0xffff
 * nor, together with the third, zero), log it as hex and mark the sensor
 * valid; otherwise clear the stored reading and mark it invalid. */
#include <stdint.h>
#include "firmware/reconstructed/include/sr08_radio_configuration.h"
#include "firmware/reconstructed/include/sr08_time_event_queue.h"

void format_byte_span_spaced_hexadecimal_line_at_b178(const uint8_t *bytes, uint32_t length);

/* Distinct integer-offset annotations preserve the OEM root-relative
 * accesses and keep the two conditional third-value reads separate. */
#define queue_half(offset, tag) (*(volatile uint16_t *)((volatile uint8_t *)queue + __builtin_annotation(offset, tag)))
#define state_half(offset, tag) (*(volatile uint16_t *)((volatile uint8_t *)state + __builtin_annotation(offset, tag)))

void save_valid_measurement_range_clear_stale_at_8500(void)
{
    volatile struct time_event_queue_state *queue = &time_event_queue_state;
    volatile struct sr08_radio_configuration_state *state = (volatile struct sr08_radio_configuration_state *)&device_control_state;

    uint32_t one = __builtin_annotation(1,"sensor readiness flag");
    if (queue->sensor_reading_status == 1) {
        uint16_t first = queue_half(0x40, "first input");
        uint16_t third;
        if (first == 0) {
            third = queue_half(0x44, "third zero input");
            if (third == 0)
                goto invalid;
        } else {
            if (first == 0xffff)
                goto invalid;
            third = queue_half(0x44, "third nonzero input");
        }
        state_half(0x42, "status output") = first;
        state_half(0x46, "value_2 output") = third;
        state_half(0x44, "value_1 output") = queue_half(0x42, "value_1 input");
        state_half(0x48, "value_3 output") = queue_half(0x46, "value_3 input");
        format_byte_span_spaced_hexadecimal_line_at_b178((const uint8_t *)&queue->sensor_reading,
                                                          sizeof(queue->sensor_reading));
        state->sensor_state_19 = one;
        state->sensor_state_1a = 0;
        return;
    }
invalid:
    state_half(0x42, "status output") = 0;
    state_half(0x48, "value_3 output") = 0;
    typedef uint32_t alias_word __attribute__((may_alias));
    *(volatile alias_word *)__builtin_assume_aligned((volatile uint8_t *)state + 0x44, 4) = 0;
    state->sensor_state_1a = 1;
    state->sensor_state_19 = 0;
}
