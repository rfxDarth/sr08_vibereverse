/* Recovered behavior:
 * For action 2, initialize paired sensor state and write its profile; for actions 0x400, 0x800,
 * and 0x1000, update the selected sensor fields; otherwise apply the default profile and
 * initialize shared state when needed.
 */
#include <stdint.h>
#include <string.h>
#include "firmware/reconstructed/include/sr08_radio_configuration.h"

extern uint8_t sensor_field_state[];
extern uint8_t device_settings[];
extern void clear_cached_sensor_fields_refresh_calibration_at_8f48(uint32_t mode, uint8_t calibration_setting);
extern void write_shared_sensor_profile_fields(uint8_t profile_byte_0, uint8_t profile_byte_1, uint8_t profile_byte_2,
                                               uint32_t profile_word_04, uint32_t profile_word_08,
                                               uint8_t profile_byte_0c);
extern void copy_incoming_fields_to_shared_command_record(uint16_t first, uint32_t second, uint32_t third,
                                                          uint16_t fourth, uint16_t fifth, uint16_t sixth,
                                                          uint16_t seventh, uint16_t eighth, uint8_t ninth,
                                                          uint8_t tenth, uint8_t eleventh);
extern uint32_t retry_radio_command_until_status_0x27(void);
extern uint32_t initialize_sensor_processing_state(void);
extern void initialize_shared_control_state_tables(void);
extern void dispatch_radio_state_action_block_at_86b8(uint32_t);
extern void six_way_command_dispatch_default_branch_at_8040(uint32_t);
extern int write_indexed_timing_record(int slot_index, uint32_t timing_value, uint32_t data_word);
extern void dispatch_radio_and_sensor_state_events_block_at_9104(void);

__attribute__((used, section(".text.apply_sensor_control_action"))) void
apply_sensor_control_action(uint32_t action_code) {
    memset(sensor_field_state + 0x40, 0, 40);
    if (action_code == 0x400 || action_code == 0x1000 || action_code == 0x800) {
        clear_cached_sensor_fields_refresh_calibration_at_8f48(action_code, 7);
        return;
    }
    if (action_code == 2) {
        clear_cached_sensor_fields_refresh_calibration_at_8f48(2, 1);
        write_shared_sensor_profile_fields(30, 40, 50, 0x21340, 50000, 150);
        copy_incoming_fields_to_shared_command_record(32, 5000, 5000, 1500, 150, 500, 200, 200, 1, 37, 3);
        if (retry_radio_command_until_status_0x27() && device_control_state.reserved_1c_to_1f[2] == 0 &&
            initialize_sensor_processing_state()) {
            ((volatile uint8_t *)device_settings)[4] = 2;
            ((volatile uint8_t *)device_settings)[3] = 2;
            dispatch_radio_state_action_block_at_86b8(0);
        }
    } else {
        clear_cached_sensor_fields_refresh_calibration_at_8f48(action_code, 0);
        if (device_control_state.reserved_14_to_18[3] == 0) {
            initialize_shared_control_state_tables();
            device_settings[0] = 2;
            device_settings[1] = 2;
            six_way_command_dispatch_default_branch_at_8040(0);
        }
        if (action_code - 1 > 1)
            return;
    }
    write_indexed_timing_record(0, 40, (uint32_t)dispatch_radio_and_sensor_state_events_block_at_9104);
}
