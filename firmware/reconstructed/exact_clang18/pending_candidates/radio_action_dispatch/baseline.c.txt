/* Radio-state action dispatcher: action indices zero to five configure the
 * command record, rerun the radio profile or scan, or load one of two sensor
 * field configurations and publish it. Other indices do nothing. */
#include <stdint.h>

struct sensor_field_config {
    uint8_t enabled;
    uint8_t reserved_01;
    uint16_t value_02;
    uint32_t value_04;
    uint16_t value_08;
    uint16_t value_0a;
    uint16_t value_0c;
    uint8_t value_0e;
    uint8_t reserved_0f;
    uint32_t value_10;
    uint32_t value_14;
    uint32_t value_18;
    uint8_t value_1c;
    uint8_t reserved_1d;
    uint16_t value_1e;
};

struct sensor_field_state {
    uint8_t reserved_00_to_1f[0x20];
    struct sensor_field_config config;
};

struct radio_action_state {
    uint8_t reserved_00_to_13[0x14];
    uint8_t field_publish_active;
    uint8_t command_record_active;
    uint8_t reserved_16_to_1d[8];
    uint8_t radio_profile_active;
    uint8_t field_config_active;
};

extern struct sensor_field_state sensor_field_state;
extern volatile struct radio_action_state device_control_state;

extern void configure_command_record(uint32_t mode);
extern void configure_radio_state_from_profile(void);
extern void initialize_radio_scan_and_mark_state_ready(void);
extern void clear_radio_flag_and_configure_state(void);
extern void configure_radio_profile_and_mark_state_ready(uint32_t mode_selector);
extern void publish_enabled_sensor_field_trigger_periodic_at_8774(void);

void dispatch_radio_state_action_block_at_86b8(uint32_t action)
{
    struct sensor_field_config *config = &sensor_field_state.config;

    switch (action) {
    case 0:
        configure_command_record(2);
        device_control_state.radio_profile_active = 1;
        device_control_state.command_record_active = 1;
        break;
    case 1:
        configure_radio_state_from_profile();
        device_control_state.radio_profile_active = 0;
        break;
    case 2:
        initialize_radio_scan_and_mark_state_ready();
        device_control_state.radio_profile_active = 0;
        break;
    case 3:
        config->enabled = 15;
        config->value_0e = 5;
        config->value_0c = 16;
        config->value_04 = 0;
        config->value_0a = 4;
        config->value_08 = 0;
        config->value_02 = 0;
        config->value_10 = 0;
        config->value_14 = 0;
        config->value_18 = 0;
        config->value_1c = 0;
        config->value_1e = 0x9625;
        clear_radio_flag_and_configure_state();
        publish_enabled_sensor_field_trigger_periodic_at_8774();
        device_control_state.field_config_active = 1;
        device_control_state.field_publish_active = 1;
        break;
    case 5:
        config->enabled = 2;
        config->value_1c = 2;
        *(uint8_t *)&config->value_02 = 0;
        clear_radio_flag_and_configure_state();
        publish_enabled_sensor_field_trigger_periodic_at_8774();
        device_control_state.field_config_active = 1;
        device_control_state.field_publish_active = 1;
        break;
    case 4:
        device_control_state.field_publish_active = 0;
        configure_radio_profile_and_mark_state_ready(1);
        device_control_state.field_config_active = 0;
        break;
    }
}
