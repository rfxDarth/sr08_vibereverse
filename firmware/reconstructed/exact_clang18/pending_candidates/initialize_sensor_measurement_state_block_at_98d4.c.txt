static inline __attribute__((always_inline)) void wkupct_enable_irq(uint32_t select, uint32_t polarity, uint16_t events_num, uint16_t deb_time)
{
    SetBits16(CLK_PER_REG, 0x0010, 1); /* WAKEUPCT_ENABLE */
    *(void (*volatile *)(void))((uint8_t *)&merged_globals_5650 + 72) = conditionally_initialize_peripheral_and_schedule_service;
    SetWord16(WKUP_COMPARE_REG, events_num);
    volatile uint16_t *sel = (volatile uint16_t *)WKUP_SELECT_P0_REG;
    for (uint32_t i = 0; i != 32; i += 8, sel++) {
        *sel = (*sel & ~0xff) + (uint8_t)(select >> i);
        sel[-4] = (sel[-4] & ~0xff) + (uint8_t)(polarity >> i);
    }
    REG16(WKUP_SELECT_P3_REG) &= 0xfcff;
    REG16(WKUP_POL_P3_REG) &= 0xfcff;
    SetWord16(WKUP_RESET_IRQ_REG, 1);
    SetWord16(WKUP_RESET_CNTR_REG, 0);
    SetBits16(WKUP_CTRL_REG, 0x3f, deb_time);
    SetBits16(WKUP_CTRL_REG, 0x80, 1);
    REG32(NVIC_IPR4) = (REG32(NVIC_IPR4) | 0x80) & ~0x7f;
    REG32(NVIC_ISER) = 1u << 16;
}

/* sdk/platform/arch/main/arch_system.c */
static inline __attribute__((always_inline)) void set_xtal16m_trim_value(uint16_t trim_value)
{
    G5650(uint16_t, 0x3a) = trim_value; /* clk_freq_trim_reg_value */
    SetBits16(CLK_16M_REG, 0x0001, 1);
    arch_asm_delay_us(13);
    SetBits16(CLK_CTRL_REG, 0x0003, 1);
    while ((GetWord16(CLK_CTRL_REG) & 0x0040) == 0)
        ;
    SetWord16(CLK_FREQ_TRIM_REG, trim_value);
    arch_asm_delay_us(1000);
    SetBits16(CLK_CTRL_REG, 0x0003, 0);
    while ((GetWord16(CLK_CTRL_REG) & 0x0080) == 0)
        ;
}

static inline __attribute__((always_inline)) void init_pwr_and_clk_ble(void)
{
    SetBits16(CLK_RADIO_REG, 0x0030, 0);  /* BLE_DIV */
    SetBits16(CLK_RADIO_REG, 0x0080, 1);  /* BLE_ENABLE */
    SetBits16(CLK_RADIO_REG, 0x0003, 1);  /* RFCU_DIV */
    SetBits16(CLK_RADIO_REG, 0x0008, 1);  /* RFCU_ENABLE */
    SetBits16(CLK_32K_REG, 0x0080, 1);    /* RC32K_ENABLE */
    SetBits16(SYS_CTRL_REG, 0x0008, 0);   /* CLK32_SOURCE */
    SetBits16(CLK_RADIO_REG, 0x0040, 1);  /* BLE_LP_RESET */
    SetBits16(PMU_CTRL_REG, 0x0004, 0);   /* RADIO_SLEEP */
    while (!(GetWord16(SYS_STAT_REG) & 0x0002))
        ;
    SetBits16(CLK_RCX20K_REG, 0x00f0, 0xb); /* RCX20K_NTC */
    SetBits16(CLK_RCX20K_REG, 0x0300, 0);   /* RCX20K_BIAS */
    SetBits16(CLK_RCX20K_REG, 0x000f, 0);   /* RCX20K_TRIM */
    SetBits16(CLK_RCX20K_REG, 0x0400, 1);   /* RCX20K_LOWF */
    SetBits16(CLK_RCX20K_REG, 0x0800, 1);   /* RCX20K_ENABLE */
    SetBits16(CLK_RCX20K_REG, 0x1000, 1);   /* RCX20K_SELECT */
    SetBits16(SYS_CTRL_REG, 0x0008, 0);
    SetBits16(CLK_32K_REG, 0x0001, 0);      /* XTAL32K_ENABLE */
    SetBits16(CLK_32K_REG, 0x0080, 0);
    SetBits16(CLK_RADIO_REG, 0x0040, 0);
    if (GetWord16(ANA_STATUS_REG) & 0x0200)
        SetWord16(DCDC_CTRL3_REG, 0x5);
    SetBits32(BLE_RWBLECNTL_REG, 0x00000100, 0);
    SetBits32(BLE_CNTL2_REG, 0x00007e00, 16);
    SetBits32(BLE_CNTL2_REG, 0x00200000, 1);
}

    SetWord32(BLE_SAMPLECLK_REG, 1);
    while (GetWord32(BLE_SAMPLECLK_REG))
        ;
    return GetWord32(BLE_BASETIMECNT_REG);
}

/* sdk/platform/arch/main/arch_system.c conditionally_run_radio_cals() */
static inline __attribute__((always_inline)) void conditionally_run_radio_cals(void)
{
    uint32_t current_time = ble_basetime();
    uint32_t last = G5650(uint32_t, 68);
    uint16_t count;
    if (current_time < last) {
        G5650(uint32_t, 68) = 0;
        last = 0;
    }
    if (G5650(uint8_t, 1) == 1) {
        G5650(uint32_t, 68) = current_time;
        G5650(uint8_t, 1) = 0;
        count = get_rc16m_count();
    } else {
        if (current_time - last <= 0xc7f)
            return;
        G5650(uint32_t, 68) = current_time;
        count = get_rc16m_count();
        uint16_t previous = G5650(uint16_t, 56);
        uint16_t diff = previous >= count ? previous - count : count - previous;
        if (diff < 24)
            return;
    }
    G5650(uint16_t, 56) = count;
    if (rf_calibration_585() == 0)
        G5650(uint8_t, 1) = 1;
}

    ((volatile uint32_t *)ring)[2] = size;
    ((volatile uint32_t *)ring)[3] = width;
    ((void *volatile *)ring)[4] = buffer;
}

static inline __attribute__((always_inline)) void vendor_powered_tick(void)
{
    if (G5650(uint8_t, 13) == 0)
        vendor_first_powered_init();
    uint32_t delay;
    if (sr08_timing_records[0] == 1 || G5650(uint8_t, 52) != 0)
        delay = 1;
    else if (G5650(uint8_t, 40) == 0)
        delay = 100;
    else
        delay = 11;
    ke_timer_set(APP_TIMER, TASK_APP, delay);
    uint8_t mode = G5650(uint8_t, 15);
    if (!(GetWord16(P1_DATA_REG) & 0x20)) {
        if (mode == 2) {
            arch_printf(uart_format_string_and_decimal, uart_newline_string, 335);
            uint32_t event = 1;
            enqueue_ring_record_if_space(&merged_globals_574c, &event);
            G5650(uint8_t, 15) = 0;
        }
    } else {
        switch (mode) {
        case 0:
            G5650(uint8_t, 15) = 2;
            G5650(uint32_t, 108) = ble_basetime();
            /* fall through */
        case 2: {
            uint32_t now = ble_basetime();
            uint32_t last = G5650(uint32_t, 108);
            if ((now >= last ? now : now + 0x5000000) - last >= 0x1450) {
                arch_printf(uart_format_string_and_decimal, uart_newline_string, 325);
                uint32_t event = 8;
                enqueue_ring_record_if_space(&merged_globals_574c, &event);
                G5650(uint8_t, 15) = 4;
            }
            break;
        }
        }
    }
    wkupct_enable_irq(0x2000, 0x2000, 1, 0);
}

static inline __attribute__((always_inline)) void vendor_process_events(void)
{
    uint32_t event = 0;
    while (peek_ring_buffer_entry(&merged_globals_574c, &event)) {
        advance_queue_tail_if_nonempty(&merged_globals_574c);
        arch_printf("k=%d\n", event);
        if (event == 8 || event == 1)
            G5650(uint16_t, 64) = 6;
        uint8_t current = G5650(uint8_t, 12);
        for (uint32_t i = 0; i < 36; i++) {
            const struct display_state_descriptor *desc = display_state_descriptors[i];
            if (desc->state != current)
                continue;
            const struct display_event_handler *h = desc->handlers;
            for (uint32_t n = desc->handler_count; n; n--, h++) {
                if (h->mask & event) {
                    if (h->handler)
                        h->handler(event);
                    if (h->next_state != 0xffff)
                        switch_display_state_and_run_hooks(h->next_state);
                    break;
                }
            }
            break;
        }
        event = 0;
    }
    for (uint32_t i = 0; i != 192; i += 16) {
        struct timing_record *rec = (struct timing_record *)(sr08_timing_records + i);
        if (rec->active != 1)
            continue;
        uint32_t now = get_ble_base_time_ms();
        uint32_t elapsed = (now >= rec->last ? now : now + 0x5000000) - rec->last;
        if (elapsed >= rec->interval) {
            rec->last = now;
            rec->callback();
            if (rec->repeat == 0)
                sr08_timing_records[i] = 0;
        }
    }
}

/* sdk/platform/core_modules/rwip/src/rwip.c rwip_sleep() and arch_main.c arch_goto_sleep(), RCX20 build. */
static inline __attribute__((always_inline)) void sleep_and_resume(void)
{
    __asm__ volatile("cpsid i");
    uint32_t sleep_duration = rwip_env_bytes[4] == 2 ? 0 : rom_cfg_table[10];
    if (GetWord16(GP_CONTROL_REG) & 1)
        goto active;
    if (!ke_sleep_check())
        goto active;
    if (!rwip_env_bytes[3])
        goto idle;
    if (!(GetWord16(SYS_STAT_REG) & 1)) {
        if (*(volatile uint16_t *)rwip_env_bytes != 0)
            goto idle;
        if (ke_env_queue_timer_first != 0)
            sleep_duration = rom_cfg_table[10];
        if (!ke_timer_sleep_check(&sleep_duration, rwip_env_bytes[2]))
            goto idle;
        if (GetWord32(BLE_INTSTAT_REG) & 0x40)
            goto idle;
        if (!ea_sleep_check(&sleep_duration, rwip_env_bytes[2]))
            goto idle;
        if (sleep_duration && GetWord32(BLE_FINETIMECNT_REG) - 312 <= 218)
            sleep_duration--;
        uint32_t lp = G56D0(uint32_t, 8);
        uint32_t twirq_set_value = (3357 * lp + 500000) / 1000000;
        uint32_t twirq_reset_value = (2900 * lp + 500000) / 1000000;
        SetWord32(BLE_ENBPRESET_REG, (GetWord32(BLE_ENBPRESET_REG) & ~0x3ffu) | (twirq_set_value - twirq_reset_value));
        uint32_t twext_value = (sleep_env_slp_state == 2 ? 35 : 18) + twirq_set_value;
        SetWord32(BLE_ENBPRESET_REG, (GetWord32(BLE_ENBPRESET_REG) & 0xffe003ff) | (twext_value << 10));
        SetWord32(BLE_ENBPRESET_REG, (GetWord32(BLE_ENBPRESET_REG) & 0x1fffff) + (twirq_set_value << 21));
        if (sleep_duration >= 1000000) {
            wdg_reload(200);
            SetWord16(RESET_FREEZE_REG, 8);
            while (1)
                ;
        }
        uint64_t scaled = (uint64_t)(G56D0(uint32_t, 80) | ((uint64_t)G56D0(uint32_t, 84) << 32)) * sleep_duration;
        uint32_t sleep_lp_cycles = (uint32_t)(scaled >> 20) - 1;
        if (sleep_lp_cycles == twext_value + 1)
            sleep_lp_cycles = twext_value;
        lld_sleep_enter(sleep_lp_cycles, rwip_env_bytes[4] != 0);
        (*(void (**)(void))(0x07FD7FB4 + 44))();
        while (!(GetWord32(BLE_DEEPSLCNTL_REG) & 0x8000))
            ;
        while (!(GetWord32(BLE_CNTL2_REG) & 0x100))
            ;
        SetWord32(BLE_CNTL2_REG, GetWord32(BLE_CNTL2_REG) & ~0x80u);
        uint16_t bandgap = GetWord16(BANDGAP_REG_SLP);
        if (bandgap <= 69) {
            SetWord16(BANDGAP_REG_SLP, 70);
            G5650(uint32_t, 80) = 70 - bandgap;
        }
        ble_regs_push();
        G5650(uint32_t, 84) = GetWord32(BLE_CNTL2_RET_REG);
        while (!(GetWord32(BLE_CNTL2_REG) & 0x80))
            ;
        SetBits16(CLK_RADIO_REG, 0x0080, 0);
    }
    /* arch_goto_sleep() */
    SetBits16(PMU_CTRL_REG, 0x0004, 1);
    REG32(SCB_SCR) |= 4;
    SetBits16(SYS_CTRL_REG, 0x0020, 0);
    SetBits16(PMU_CTRL_REG, 0x0002, 1);
    if (sleep_env_slp_state == 1) {
        SetBits16(SYS_CTRL_REG, 0x0010, 0);
    } else {
        SetBits16(SYS_CTRL_REG, 0x0f00, 0);
        hw_otpc_init_and_power_on();
        SetBits16(SYS_CTRL_REG, 0x0010, 1);
        SetWord32(OTPC_NWORDS_REG, ((0x000104c0 + 0x07fc0130 + 0xf8040003) >> 2) - 1);
        SetBits16(CLK_AMBA_REG, 0x0080, 0);
    }
    REG16(PMU_CTRL_REG) |= 0x0f00;
    SetBits16(CLK_16M_REG, 0x0100, 0);
    SetBits16(CLK_16M_REG, 0x0001, 1);
    arch_asm_delay_us(13);
    SetBits16(CLK_CTRL_REG, 0x0003, 1);
    while ((GetWord16(CLK_CTRL_REG) & 0x0040) == 0)
        ;
    SetWord16(CLK_FREQ_TRIM_REG, 0);
    if (GetWord16(SYS_STAT_REG) & 0x20)
        SetWord16(SET_FREEZE_REG, 8);
    __asm__ volatile("wfi");
    if (GetWord16(SYS_STAT_REG) & 0x20)
        SetWord16(RESET_FREEZE_REG, 8);
    REG32(SCB_SCR) &= ~4u;
    goto active;
idle:
    __asm__ volatile("wfi");
active:
    __asm__ volatile("cpsie i");
}

__attribute__((used, section(".text.initialize_sensor_measurement_state_block_at_98d4"))) void
initialize_sensor_measurement_state_block_at_98d4(void)
{
    system_init();
    while (1) {
        if ((GetWord16(CLK_RADIO_REG) & 0x80) && !(GetWord32(BLE_DEEPSLCNTL_REG) & 0x8000) &&
            !(*(volatile uint16_t *)rwip_env_bytes & 1)) {
            rwip_schedule();
            if (G5650(uint8_t, 3) == BLE_EVT_END) {
                wait_sensor_readiness_combine_counter_words_at_b7d4(); /* rcx20_read_freq() */
                uint32_t sleep_duration = 0;
                if (ea_sleep_check(&sleep_duration, 4))
                    conditionally_run_radio_cals();
            }
            vendor_powered_tick();
            vendor_process_events();
            if (G5650(uint32_t, 120) != 0) {
                __asm__ volatile("wfi");
                wdg_reload(200);
                continue;
            }
        }
        sleep_and_resume();
        wdg_reload(200);
    }
}
